"""앞 판정에서 어디서도 안 잡힌 뮤턴트 10개 + 이번 수정 자리 뮤턴트를 A-117 판에 다시 건다. 픽스처 test 만, 케이스마다 파일에 쓴다."""
import json, os, pathlib, re, shutil, subprocess, sys
E = pathlib.Path(sys.argv[1]); WORK = pathlib.Path(sys.argv[2]); OUT = pathlib.Path(sys.argv[3])
S = E / "tests" / "structure"
SRC = (S / "rule_l_checks.py").read_text(encoding="utf-8")
M = [
 ("Op-float-cuts", 'if isinstance(fn, ast.Name) and fn.id in ("float", "int", "str", "bool", "dict", "tuple", "list", "set", "frozenset", "sorted", "reversed", "getattr") and expr.args:', 'if isinstance(fn, ast.Name) and fn.id in ("str", "bool", "dict", "tuple", "list", "set", "frozenset", "sorted", "reversed", "getattr") and expr.args:'),
 ("Op-output-exempt-off", 'op = _OpCopy(eng, reach, (kind, field), output_sids, writer_file)', 'op = _OpCopy(eng, reach, (kind, field), set(), writer_file)'),
 ("Op-writer-exempt-off", 'and eng.modules[eng.scopes[sid].module] != writer_file and not _under(sid, exempt_sids)]', 'and not _under(sid, exempt_sids)]'),
 ("Op-reach-restored", 'self.cands = [sid for sid in sorted(eng.scopes) if not sid.endswith(":<module>")', 'self.cands = [sid for sid in sorted(reach) if sid in eng.scopes and not sid.endswith(":<module>")'),
 ("L4-exempt-off", 'if slot in loads or slot in strings or any(key == e or key.endswith("." + e) for e in exempt):', 'if slot in loads or slot in strings:'),
 ("L3-unresolved-reads-ok", 'out.append(Violation(("L-3", label + ".<reads>"), "reads_unverifiable", "reads= 를 상수로 풀지 못했다"))', 'pass'),
 ("Table-wt-off", 'self.unresolved.append((("L-1", name, eng.modules[mod]), "writer_table_unresolved", "WriterTable 상수 해석 실패: %s" % exc))', 'pass'),
 ("Entry-missing-off", 'out.append(Violation(("L-1", "<entry>", spec), "entry_point_missing", "진입점 함수가 엔진에 없다"))', 'pass'),
 ("Graph-shadow-ignored", 'elif node.id not in loc:\n                            targets = self._targets_of', 'elif True:\n                            targets = self._targets_of'),
 ("Read-kindfilter-off", 'if kind_vals is not None and not set(kind_vals) <= kinds:\n                    continue', 'if False:\n                    continue'),
 # 이번 수정 자리
 ("L3-alias-off", 'tainted.add(target.id)\n                                    grew = True', 'grew = False'),
 ("Field-prefix-off", 'if head[1:] in (attr, "?"):', 'if True:'),
 ("Cap-drop-deep", 'return "@?|" + lab.rpartition("|")[2]', 'return "@none|" + lab.rpartition("|")[2]'),
 ("Global-sink-off", 'if isinstance(target, ast.Attribute) or (is_global and (target is not base or base.id in globals_)):', 'if isinstance(target, ast.Attribute):'),
 ("Gupd-off", 'sink(set(got_all))                       # 모듈 전역 컨테이너에 담는다', 'pass'),
 ("Serial-off", 'if isinstance(fn, ast.Attribute) and fn.attr in SERIALIZERS and got_src:\n                        sink(got_src)', 'if False:\n                        sink(got_src)'),
 ("Unknown-meth-off", 'sink(got_src)                                # 엔진 안에 정의가 없는 메서드: 추적 불능', 'pass'),
 ("Pin-unwrap-off", 'if (unwrap and isinstance(expr, ast.Call)', 'if (False and isinstance(expr, ast.Call)'),
 ("Pin-builtin-unwrap", 'isinstance(expr.func, ast.Name) and expr.func.id not in BUILTIN_WRAP', 'isinstance(expr.func, ast.Name)'),
 ("Pin-diode-unwrap", '{_pin_token(node.left), _pin_token(node.comparators[0])}', '{_pin_token(node.left, True), _pin_token(node.comparators[0], True)}'),
]
d = WORK / "m2"
if d.exists():
    shutil.rmtree(d)
d.mkdir(parents=True)
for f in ("rule_l_tables.py", "rule_l_debt.py", "test_rule_l_reject_fixtures.py"):
    shutil.copy(S / f, d / f)
shutil.copytree(S / "fixtures" / "rule_l", d / "fixtures" / "rule_l")
env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1", PYTHONIOENCODING="utf-8", OPENBLAS_NUM_THREADS="1")
with open(OUT, "w", encoding="utf-8") as fh:
    for name, a, b in M:
        n = SRC.count(a)
        if n != 1:
            row = {"name": name, "error": "occurrences=%d" % n}
        else:
            (d / "rule_l_checks.py").write_text(SRC.replace(a, b), encoding="utf-8")
            r = subprocess.run([sys.executable, "-B", "-m", "pytest", "test_rule_l_reject_fixtures.py", "-q", "--tb=no", "-p", "no:cacheprovider", "--rootdir", str(d), "-c", os.devnull],
                               cwd=d, env=env, capture_output=True, text=True, encoding="utf-8", errors="replace")
            fails = re.findall(r"^(?:FAILED|ERROR) \S*?(?:::)?(\S+)", r.stdout, re.M)
            tail = r.stdout.strip().splitlines()[-1] if r.stdout.strip() else r.stderr[-160:]
            row = {"name": name, "killed": r.returncode != 0, "n_fail": len(fails), "fails": fails[:3], "tail": tail[:80]}
        fh.write(json.dumps(row, ensure_ascii=False) + "\n"); fh.flush()
        print(row)
