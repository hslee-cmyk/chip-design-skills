"""살아남은 뮤턴트를 실제 엔진에 돌려 위반 집합 차이(baseline 기준 제안과 대조)를 잰다. 케이스마다 파일에 쓴다."""
import importlib.util, json, pathlib, re, sys
E = pathlib.Path(sys.argv[1]); OUT = pathlib.Path(sys.argv[2]); IDS = [int(x) for x in sys.argv[3].split(",")]
scripts = pathlib.Path(__file__).parent
src = (scripts / "mutate.py").read_text(encoding="utf-8")
# M 표만 꺼낸다
body = src[src.index("M = ["):src.index("start = int")]
ns = {}
exec(body, ns)
M = ns["M"]
S = E / "tests" / "structure"
sys.path.insert(0, str(S))
from rule_l_tables import PARAMS
SRC = (S / "rule_l_checks.py").read_text(encoding="utf-8")
base_v = json.loads(pathlib.Path(sys.argv[4]).read_text(encoding="utf-8"))
base = {tuple(v["key"]) for v in base_v}
tmp = pathlib.Path(sys.argv[5]); tmp.mkdir(exist_ok=True)
with open(OUT, "a", encoding="utf-8") as fh:
    for i in IDS:
        name, a, b = M[i]
        p = tmp / ("mut_%d.py" % i)
        p.write_text(SRC.replace(a, b), encoding="utf-8")
        spec = importlib.util.spec_from_file_location("mut_%d" % i, p)
        mod = importlib.util.module_from_spec(spec); sys.modules[spec.name] = mod
        spec.loader.exec_module(mod)
        try:
            got = {v.key for v in mod.check_rule_l(E, PARAMS)}
            row = {"i": i, "name": name, "lost": sorted(base - got)[:8], "n_lost": len(base - got), "gained": sorted(got - base)[:8], "n_gained": len(got - base)}
        except Exception as exc:
            row = {"i": i, "name": name, "exc": repr(exc)[:200]}
        fh.write(json.dumps(row, ensure_ascii=False) + "\n"); fh.flush()
        print(row["i"], row["name"], row.get("n_lost"), row.get("n_gained"), row.get("exc", ""))
