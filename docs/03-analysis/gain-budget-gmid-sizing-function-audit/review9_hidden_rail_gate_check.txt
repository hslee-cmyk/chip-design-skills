  ngcc_3stage_ff                 숨은넷=[] DC고립=[] 거부=no []
  nmcnr_amplifier                숨은넷=[] DC고립=[] 거부=no []
  pushpull_follower_buffer       숨은넷=['agnd'] DC고립=['agnd'] 거부=YES ['agnd']
  sf_pushpull_output_opamp       숨은넷=['agnd'] DC고립=['agnd'] 거부=YES ['agnd']
  two_stage_ota                  숨은넷=[] DC고립=[] 거부=no []
  two_stage_ota_sf               숨은넷=[] DC고립=[] 거부=no []

  pushpull_follower_buffer       supply_nets=['vdd','vss'](agnd 누락) -> 거부=YES ['agnd']
  sf_pushpull_output_opamp       supply_nets=['vdd','vss'](agnd 누락) -> 거부=YES ['agnd']
